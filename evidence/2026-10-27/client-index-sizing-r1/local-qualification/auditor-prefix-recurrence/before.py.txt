1    #!/usr/bin/env python3
"""Replay every received common-subset command from original CSV, independently of Java."""
import argparse
import csv
import hashlib
import json
from pathlib import Path

BASE, QUOTE, ACCOUNTS = 1_000_000_000, 1_000_000_000_000, 16


def require(ok, reason):
    if not ok:
        raise ValueError(reason)


def quantiles(values):
    if not values:
        return {}
    a = sorted(values)
    n = len(a)
    return {"count": n, "p50": a[(n * 50 + 99) // 100 - 1],
            "p99": a[(n * 99 + 99) // 100 - 1],
            "p999": a[(n * 999 + 999) // 1000 - 1], "max": a[-1]}


class Book:
    def __init__(self):
        self.live = {}
        self.levels = {True: {}, False: {}}
        self.base = [BASE] * (ACCOUNTS + 1)
        self.quote = [QUOTE] * (ACCOUNTS + 1)
        self.used = set()

    def remove(self, order):
        level = self.levels[order["buy"]][order["price"]]
        del level[order["id"]]
        if not level:
            del self.levels[order["buy"]][order["price"]]
        del self.live[order["id"]]

    def apply(self, row):
        kind = row["kind"]
        require(kind in {"GTC", "IOC", "CANCEL"}, "unsupported operation")
        ident, account, price, quantity = (int(row[k]) for k in ["id", "account", "price", "quantity"])
        require(row["buy"] in {"true", "false"}, "invalid side")
        buy = row["buy"] == "true"
        require(0 < account <= ACCOUNTS and ident > 0 and price > 0 and quantity > 0, "invalid command")
        if kind == "CANCEL":
            order = self.live.get(ident)
            if order is None or order["account"] != account:
                return False, 0, []
            self.remove(order)
            return True, order["remaining"], []
        require(ident not in self.used, "duplicate original place ID")
        self.used.add(ident)
        remaining, fills = quantity, []
        levels = self.levels[not buy]
        while remaining and levels:
            best = min(levels) if buy else max(levels)
            if (buy and best > price) or (not buy and best < price):
                break
            maker = next(iter(levels[best].values()))
            require(maker["account"] != account, "unsupported self trade")
            traded = min(remaining, maker["remaining"])
            buyer, seller = (account, maker["account"]) if buy else (maker["account"], account)
            self.base[buyer] += traded
            self.base[seller] -= traded
            self.quote[buyer] -= traded * best
            self.quote[seller] += traded * best
            fills.append((maker["id"], ident, best, traded))
            maker["remaining"] -= traded
            remaining -= traded
            if not maker["remaining"]:
                self.remove(maker)
        if remaining and kind == "GTC":
            order = dict(id=ident, account=account, buy=buy, price=price, remaining=remaining)
            self.live[ident] = order
            self.levels[buy].setdefault(price, {})[ident] = order
        return True, remaining, fills

    def state(self):
        br, qr = [0] * (ACCOUNTS + 1), [0] * (ACCOUNTS + 1)
        orders = sorted(self.live.values(), key=lambda o: o["id"])
        for order in orders:
            a = order["account"]
            if order["buy"]:
                qr[a] += order["price"] * order["remaining"]
            else:
                br[a] += order["remaining"]
        accounts = []
        for a in range(1, ACCOUNTS + 1):
            require(self.base[a] >= br[a] and self.quote[a] >= qr[a], "negative funds")
            accounts.append(dict(account=a, baseAvailable=self.base[a] - br[a], baseReserved=br[a],
                                 quoteAvailable=self.quote[a] - qr[a], quoteReserved=qr[a]))
        require(sum(self.base[1:]) == ACCOUNTS * BASE and sum(self.quote[1:]) == ACCOUNTS * QUOTE,
                "total asset conservation")
        return dict(accounts=accounts, orders=orders)


def sha(path):
    h = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            h.update(block)
    return h.hexdigest()


def audit(root):
    root = Path(root)
    config = json.loads((root / "configuration.json").read_text())
    result = json.loads((root / "result.json").read_text())
    origin = json.loads((root / "origin.json").read_text())["originMonotonicNs"]
    require(result["correctnessPassed"] and config["count"] == result["planned"], "native result not complete")
    require(sha(root / "commands.csv") == result["sourceSha256"], "source hash differs")
    require(config["count"] == config["rate"] * config["seconds"] <= 1_250_000, "plan bounds")
    oracle = Book()
    setup = count = accepted = rejected = not_sent = trades = window = accepted_window = 0
    global_latency, dispatch_latency, scheduler = [], [], []
    maximum_end = 0
    with (root / "commands.csv").open() as cs, (root / "outcomes.csv").open() as rs, (root / "trades.csv").open() as ts:
        commands, replies, actual_trades = csv.DictReader(cs), csv.DictReader(rs), csv.DictReader(ts)
        for c in commands:
            if c["phase"] == "SETUP":
                require(count == 0 and int(c["index"]) == setup, "setup order")
                ok, _, fills = oracle.apply(c)
                require(ok and not fills, "setup must rest")
                setup += 1
                continue
            require(c["phase"] == "MEASURED" and setup == 1024 and int(c["index"]) == count, "original command coverage")
            r = next(replies, None)
            require(r is not None and int(r["index"]) == count, "original outcome coverage")
            planned = count * 1_000_000_000 // config["rate"]
            require(int(r["planned_offset_ns"]) == planned, "arrival clock changed")
            source, begin, end = (int(r[k]) for k in ["source_ns", "publisher_ns", "reply_ns"])
            require(source >= origin + planned, "source before original schedule")
            scheduler.append(source - origin - planned)
            if r["status"] == "NOT_SENT":
                require(begin == end == 0 and r["accepted"] == r["remaining"] == r["trades"] == "", "not-sent has fabricated outcome")
                not_sent += 1
            else:
                require(r["status"] == "REPLIED" and end >= begin >= source, "unknown or invalid time order")
                ok, remaining, fills = oracle.apply(c)
                require(r["accepted"] == str(ok).lower() and int(r["remaining"]) == remaining and int(r["trades"]) == len(fills),
                        "independent outcome differs at " + str(count))
                for ordinal, fill in enumerate(fills):
                    t = next(actual_trades, None)
                    require(t is not None and int(t["index"]) == count and int(t["ordinal"]) == ordinal,
                            "missing/disordered original trade")
                    require(tuple(int(t[k]) for k in ["maker", "taker", "price", "quantity"]) == fill, "price/time/settlement differs")
                accepted += ok
                rejected += not ok
                trades += len(fills)
                in_window = end < origin + config["seconds"] * 1_000_000_000
                window += in_window
                accepted_window += in_window and ok
                global_latency.append(end - origin - planned)
                dispatch_latency.append(end - begin)
                maximum_end = max(maximum_end, end)
            count += 1
        require(next(replies, None) is None and next(actual_trades, None) is None, "unattributed raw rows")
    require(count == config["count"] and setup == 1024, "incomplete source")
    expected = dict(planned=count, replied=accepted + rejected, accepted=accepted, businessRejected=rejected,
                    notSent=not_sent, unknown=0, trades=trades, inWindowReplied=window, inWindowAccepted=accepted_window,
                    conditionalReplyLatencyNs=quantiles(global_latency), conditionalPublisherToReplyNs=quantiles(dispatch_latency),
                    sourceLatenessNs=quantiles(scheduler), allArrivalAcceptedLatencyNs=quantiles(global_latency) if accepted == count else None,
                    postWindowDrainNs=max(0, maximum_end - origin - config["seconds"] * 1_000_000_000),
                    schedulerValid=not any(x > 20_000_000 for x in scheduler))
    expected["finiteGatePassed"] = (accepted == count and window * 100 >= count * 99 and expected["schedulerValid"])
    for key, value in expected.items():
        require(result[key] == value, "summary mismatch: " + key)
    require(oracle.state() == json.loads((root / "final-state.json").read_text()), "final money/open orders differ")
    return dict(passed=True, independentPythonBusinessReplay=True, verified=expected,
                sourceSha256=result["sourceSha256"], fullOutcomeSha256=sha(root / "outcomes.csv"),
                fullTradeSha256=sha(root / "trades.csv"), analyserSha256=sha(Path(__file__)),
                scope="Common-subset in-process API result; no replication/durability/SLA qualification")


if __name__ == "__main__":
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("root", type=Path)
    parser.add_argument("--output", type=Path, required=True)
    args = parser.parse_args()
    result = audit(args.root)
    with args.output.open("x") as stream:
        json.dump(result, stream, indent=2)
        stream.write("\n")
    print(json.dumps({"passed": result["passed"], "verified": result["verified"]}))
